class Solution {
    public int maxArea(int[] height) {
        int n = height.length;
        int ans = 0;

        int first = 0;
        int last = n - 1;

        while (first < last) {
            int h = Math.min(height[first], height[last]);
            int l = last - first;
            int temp = h * l;

            ans = Math.max(ans, temp);
            if (height[first] < height[last]) {
                first++;
            } else {
                last--;
            }
        }

        return ans;

    }
}
