class Solution {
    public List<List<Integer>> threeSum(int[] nums) {
        int n = nums.length;

        Arrays.sort(nums);
        HashSet<List<Integer>> hs = new HashSet<>();

        for (int i = 0; i < n - 1; i++) {
            int first = nums[i];
            int second = i + 1;
            int last = n - 1;

            while (second < last) {
                int sum = first + nums[second] + nums[last];
                if (sum == 0) {
                    List<Integer> temp = new ArrayList<>();
                    temp.add(first);
                    temp.add(nums[second]);
                    temp.add(nums[last]);
                    hs.add(temp);
                    second++;
                    last--;
                } else {
                    if (sum > 0) {
                        last--;
                    } else {
                        second++;
                    }
                }
            }
        }

        List<List<Integer>> ls = new ArrayList<>();
        for (List<Integer> a : hs) {
            ls.add(a);
        }

        return ls;
    }
}
