class Solution {
    public List<List<Integer>> fourSum(int[] nums, int target) {
        int n = nums.length ;
        HashSet<List<Integer>> hs = new HashSet<>() ;

        Arrays.sort(nums) ;
        for(int i = 0 ; i<n-3 ; i++){
            for(int j = i+1 ; j<n-2 ; j++){
                int start = j+1 ;
                int end = n-1 ;
                while(start<end){
                    long sum = (long)nums[i]+nums[j]+nums[start]+nums[end];
                    if(sum==target){
                        ArrayList<Integer> temp = new ArrayList<>() ;
                        temp.add(nums[i]);
                        temp.add(nums[j]);
                        temp.add(nums[start]);
                        temp.add(nums[end]);
                        hs.add(temp) ;

                        start++;
                        end--;


                    } else if (sum < target) {
                        start++;
                    } else {
                        end--;
                    }
                }
            }
        }
        
        return new ArrayList<>(hs);
    }
}
